        function getQueryParam(name) {
            const urlParams = new URLSearchParams(window.location.search);
            return urlParams.get(name);
        }

        // ================================================================
        // VOLVER AL INICIO (marianmadrid.es)
        // ================================================================
        function irAInicio() {
            const targetUrl = '/';
            try {
                if (window.parent && window.parent !== window) {
                    window.parent.postMessage({ type: 'MM_NAV', payload: { url: targetUrl } }, '*');
                }
                window.top.location.href = targetUrl;
            } catch (e) {
                window.location.href = targetUrl;
            }
        }

        // ================================================================
        // CARGA DE DATOS
        // ================================================================
        window.addEventListener('DOMContentLoaded', function () {
            let contextData = {};
            try {
                if (window.parent && window.parent.wixWindow) {
                    contextData = window.parent.wixWindow.getLightboxContext() || {};
                }
            } catch (e) {
                // Fallback: usar query params
            }

            // --- Lectura de datos del contexto ---
            const booking = contextData.booking || {};
            const service = contextData.service || {};

            const bookingIdF1 =
                booking.bookingIdF1 ||
                contextData.bookingIdF1 ||
                getQueryParam('bookingIdF1') ||
                '---';

            const bookingIdF2 =
                booking.bookingIdF2 ||
                contextData.bookingIdF2 ||
                getQueryParam('bookingIdF2') ||
                '';

            const serviceId =
                booking.serviceId ||
                service.serviceId ||

                getQueryParam('serviceId') ||
                '---';

            const tituloServicio =
                service.title ||
                contextData.tituloServicio ||
                getQueryParam('tituloServicio') ||
                '---';

            const fecha = booking.fecha || contextData.fecha || getQueryParam('fecha') || '---';
            const hora = booking.hora || contextData.hora || getQueryParam('hora') || '---';
            const horaF2 = booking.horaF2 || contextData.horaF2 || getQueryParam('horaF2') || '';
            const duracion = booking.duracion || contextData.duracion || getQueryParam('duracion') || '';
            const extras = booking.extras || contextData.extras || getQueryParam('extras') || 'Ninguno';

            const appointmentNote =
                booking.appointmentNote ||
                contextData.appointmentNote ||
                contextData.nota ||
                getQueryParam('appointmentNote') ||
                getQueryParam('nota') ||

            const rawPrecio = Number(
                booking.total ||
                contextData.total ||
                getQueryParam('precio') ||
                0
            );

            const esDual =
                Boolean(booking.bookingIdF2 || booking.isCombined) ||
                contextData.isCombined === true ||
                getQueryParam('esDual') === 'true';

            const assignedStaff =
                booking.staffName ||
                booking.resourceName ||

        // DESCARGAR TICKET (.TXT)
        // ================================================================
        function descargarTicket() {
            const d = window.bookingData;
            if (!d) return;

            const sep = '==================================================\n';
            let txt = '';
            txt += sep;
            txt += '              MARIAN MADRID SALÓN                 \n';
            txt += '         Peluquería, Estética y Belleza           \n';
            txt += '             Calle Jorge Juan, Madrid             \n';
            txt += sep;
            txt += '            TICKET DE CONFIRMACIÓN DE CITA        \n';
            txt += sep + '\n';

            txt += `SERVICIO:        ${d.tituloServicio}\n`;
            txt += `ID Servicio CMS: ${d.serviceId}\n`;
            txt += `ID Reserva F1:   ${d.id}\n`;
            if (d.idF2) txt += `ID Reserva F2:   ${d.idF2}\n`;

            txt += `Emisión Ticket:  ${new Date().toLocaleDateString('es-ES')} ${new Date().toLocaleTimeString('es-ES')}\n\n`;
            txt += sep;
            txt += '               DETALLES DEL SERVICIO              \n';
            txt += sep;
            txt += `Servicio:        ${d.servicio}\n`;
            if (d.esDual) {
                txt += `Modalidad:       Servicio Dual (Sincronizado)\n`;
                txt += `Fase 1:          ${d.fecha} a las ${d.hora}\n`;
                txt += `Fase 2:          ${d.fecha} a las ${d.horaF2 || 'Siguiente bloque'}\n`;
            } else {
                txt += `Cita:            ${d.fecha} a las ${d.hora}\n`;
            }
            txt += `Profesional:     ${d.estilistaDisplay}\n`;
            txt += `Complementos:    ${d.extras}\n`;
            if (d.duracion) txt += `Duración:        ${d.duracion} min\n`;
            if (d.appointmentNote) {
                txt += `Comentarios:     ${d.appointmentNote}\n`;
            }
            txt += sep;
            txt += '               INFORMACIÓN DE COBRO               \n';
            txt += sep;
            if (d.precio > 0) {
                const base = d.precio / 1.21;
                const iva = d.precio - base;
                txt += `Base Imponible:  ${base.toFixed(2)} €\n`;
                txt += `I.V.A. (21%):    ${iva.toFixed(2)} €\n`;
                txt += `Total Estimado:  ${d.precio.toFixed(2)} €\n`;
            } else {
                txt += 'Precio final calculado según tratamiento en salón\n';
            }
            txt += sep + '\n';
            txt += '               NOTAS IMPORTANTES                  \n';
            txt += '• Cancelaciones y modificaciones hasta 24h antes.\n';
            txt += '• Agradecemos que acuda 5 minutos antes de su cita.\n';
            txt += '• En servicios técnicos duales, el tiempo intermedio\n';
            txt += '  es estrictamente necesario para el correcto proceso.\n\n';
            txt += '         ¡Gracias por confiar en Marian Madrid!   \n';
            txt += sep;

            const blob = new Blob([txt], { type: 'text/plain;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            const cleanDate = String(d.fecha).replace(/\//g, '-');
            a.href = url;
            a.download = `MarianMadrid_Ticket_${d.tituloServicio}_${cleanDate}.txt`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        }

        // ================================================================
        // DESCARGAR AGENDA (.ICS)
        // ================================================================
        function descargarCalendario() {
            const d = window.bookingData;
            if (!d) return;

            let rawDate = String(d.fecha);
            let year, month, day;
            if (rawDate.includes('/')) {
                const parts = rawDate.split('/');
                day = parts[0].padStart(2, '0');
                month = parts[1].padStart(2, '0');
                year = parts[2];
            } else if (rawDate.includes('-')) {
                const parts = rawDate.split('-');
                year = parts[0];
                month = parts[1].padStart(2, '0');
                day = parts[2].padStart(2, '0');
            } else {
                const now = new Date();
                year = now.getFullYear();
                month = String(now.getMonth() + 1).padStart(2, '0');
                day = String(now.getDate()).padStart(2, '0');
            }

            const cleanHour = String(d.hora).replace(':', '') + '00';
            const startStamp = `${year}${month}${day}T${cleanHour}`;

            const hourInt = parseInt(String(d.hora).split(':')[0], 10) || 11;
            const minInt = parseInt(String(d.hora).split(':')[1], 10) || 0;
            const endHourStr = String(hourInt + 1).padStart(2, '0') + String(minInt).padStart(2, '0') + '00';
            const endStamp = `${year}${month}${day}T${endHourStr}`;

            const summary = `Marian Madrid — ${d.tituloServicio}`;
            const description =
                `Cita: ${d.servicio} (${d.estilistaDisplay}). Servicio ID: ${d.tituloServicio} (F1: ${d.id})` +
                (d.esDual ? ` F2: ${d.idF2 || 'siguiente'}` : '') +
                `\\nComplementos: ${d.extras}` +
                (d.duracion ? `\\nDuración: ${d.duracion} min` : '') +
                (d.appointmentNote ? `\\nComentarios: ${d.appointmentNote}` : '');

            let icsMsg =
                'BEGIN:VCALENDAR\n' +
                'VERSION:2.0\n' +

                'BEGIN:VEVENT\n' +
                'UID:' + d.id + '@marianmadrid.es\n' +

                'DTSTART:' + startStamp + '\n' +
                'DTEND:' + endStamp + '\n' +
                'SUMMARY:' + summary + '\n' +
                'DESCRIPTION:' + description + '\n' +
                'LOCATION:Calle Jorge Juan\\, Peluquería Marian Madrid\\, Madrid\n' +
                'END:VEVENT\n' +
                'END:VCALENDAR';

            const blob = new Blob([icsMsg], { type: 'text/calendar;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `MarianMadrid_${d.tituloServicio}_${year}${month}${day}.ics`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        }

"use strict";

(function () {
    const MESSAGE_VERSION = "1.0";
    const NAV_HOME = "/";
    const NAV_BOOKINGS = "/mis-reservas";
    const ALLOWED_ORIGIN_SUFFIXES = Object.freeze([
        "marianmadrid.es",
        ".wixsite.com",
        ".parastorage.com",
        ".wix.com",
        ".editorx.com"
    ]);
    let bookingData = null;

    function hostMatchesSuffix(host, suffix) {
        if (!host || !suffix) return false;
        const normalized = suffix.startsWith(".") ? suffix.slice(1) : suffix;
        return host === normalized || host.endsWith("." + normalized);
    }

    function isTrustedOrigin(origin) {
        if (typeof origin !== "string" || !origin || origin === "null") return false;
        let parsed;
        try {
            parsed = new URL(origin);
        } catch (_) {
            return false;
        }
        if (parsed.protocol !== "https:") return false;
        const host = parsed.hostname.toLowerCase();
        return ALLOWED_ORIGIN_SUFFIXES.some((suffix) => hostMatchesSuffix(host, suffix));
    }

    function resolveParentOrigin() {
        try {
            const ancestors = window.location.ancestorOrigins;
            if (ancestors && ancestors.length > 0) {
                const origin = String(ancestors[0] || "");
                if (isTrustedOrigin(origin)) return new URL(origin).origin;
            }
        } catch (_) {
            // Algunas versiones de navegador no exponen ancestorOrigins.
        }
        try {
            if (document.referrer) {
                const origin = new URL(document.referrer).origin;
                if (isTrustedOrigin(origin)) return origin;
            }
        } catch (_) {
            // Referrer ausente o no válido: no confiar en el origen.
        }
        return "";
    }

    const PARENT_ORIGIN = resolveParentOrigin();

    function postToParent(type, payload = {}) {
        if (
            !window.parent ||
            window.parent === window ||
            !PARENT_ORIGIN ||
            !isTrustedOrigin(PARENT_ORIGIN)
        ) return false;

        window.parent.postMessage({
            version: MESSAGE_VERSION,
            type,
            payload
        }, PARENT_ORIGIN);
        return true;
    }

    const byId = (id) => document.getElementById(id);
    const asText = (value, fallback = "") => {
        if (value === null || value === undefined) return fallback;
        const result = String(value).trim();
        return result || fallback;
    };
    const safeNumber = (value) => {
        const number = Number(value);
        return Number.isFinite(number) ? number : 0;
    };
    const durationMinutes = (value) => {
        if (typeof value === "number" && Number.isFinite(value)) return Math.max(0, value);
        const match = String(value ?? "").match(/\d+(?:\.\d+)?/);
        return match ? Math.max(0, Number(match[0])) : 0;
    };
    const setText = (id, value, fallback = "---") => {
        const node = byId(id);
        if (node) node.textContent = asText(value, fallback);
    };
    const escapeIcs = (value) => String(value ?? "")
        .replace(/\\/g, "\\\\")
        .replace(/\r\n|\n|\r/g, "\\n")
        .replace(/,/g, "\\,")
        .replace(/;/g, "\\;");
    const pad2 = (value) => String(value).padStart(2, "0");

    function sendReady() {
        return postToParent("MM_CONFIRMATION_READY");
    }

    function sendNavigation(url) {
        if (![NAV_HOME, NAV_BOOKINGS].includes(url)) return false;
        return postToParent("MM_NAV", { url });
    }

    function normalizePayload(raw) {
        if (!raw || typeof raw !== "object" || Array.isArray(raw)) return null;
        const dateTime = asText(raw.dateTime);
        const dateMatch = dateTime.match(/^(\d{2}\/\d{2}\/\d{4}) a las (\d{2}:\d{2})$/);
        const date = asText(raw.fecha, dateMatch ? dateMatch[1] : "");
        const time = asText(raw.hora, dateMatch ? dateMatch[2] : "");
        const id = asText(raw.bookingIdF1);
        if (!id || !date || !time) return null;

        return {
            id,
            idF2: asText(raw.bookingIdF2),
            serviceId: asText(raw.serviceId),
            tituloServicio: asText(raw.tituloServicio || raw.serviceName, "Servicio reservado"),
            servicio: asText(raw.serviceName || raw.tituloServicio, "Servicio reservado"),
            esDual: raw.isCombined === true || Boolean(asText(raw.bookingIdF2)),
            precio: Math.max(0, safeNumber(raw.total)),
            fecha: date,
            hora: time,
            horaF2: asText(raw.horaF2),
            extras: asText(raw.extras, "Ninguno"),
            duracion: durationMinutes(raw.duracion),
            appointmentNote: asText(raw.appointmentNote),
            status: asText(raw.status, "CONFIRMED")
        };
    }

    function render(data) {
        setText("txtServiceTitle", data.tituloServicio);
        setText("modalFecha", data.fecha);
        setText("modalHora", data.hora + (data.horaF2 ? " (Fase 2: " + data.horaF2 + ")" : ""));
        setText("modalDuracion", data.duracion > 0 ? data.duracion + " min" : "", "No indicada");
        setText("modalExtras", data.extras, "Ninguno");
        setText("txtPrecio", data.precio > 0
            ? new Intl.NumberFormat("es-ES", { style: "currency", currency: "EUR" }).format(data.precio)
            : "Precio en salón");

        const commentsBox = byId("commentsBox");
        const commentsText = byId("commentsText");
        if (commentsBox && commentsText) {
            commentsText.textContent = data.appointmentNote;
            commentsBox.style.display = data.appointmentNote ? "block" : "none";
        }

        const badge = byId("badgeDual");
        const dualInfo = byId("txtDualInfo");
        if (badge && dualInfo) {
            badge.style.display = data.esDual ? "block" : "none";
            dualInfo.textContent = data.esDual
                ? "Este servicio requiere dos bloques de agenda sincronizados.\nFase 1 (aplicación): " + data.hora +
                  "\nFase 2 (lavado/peinado): " + (data.horaF2 || "Hora pendiente") +
                  "\nEl intervalo intermedio forma parte del proceso técnico."
                : "";
            dualInfo.style.whiteSpace = "pre-line";
        }

        const button = byId("btnMisReservas");
        if (button) button.onclick = () => sendNavigation(NAV_BOOKINGS);
    }

    function dateParts(rawDate) {
        let match = String(rawDate || "").match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
        if (match) return { day: Number(match[1]), month: Number(match[2]), year: Number(match[3]) };
        match = String(rawDate || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
        if (match) return { year: Number(match[1]), month: Number(match[2]), day: Number(match[3]) };
        return null;
    }

    function localDateTime(data) {
        const parts = dateParts(data.fecha);
        const time = String(data.hora).match(/^(\d{2}):(\d{2})$/);
        if (!parts || !time || parts.month < 1 || parts.month > 12 ||
            parts.day < 1 || parts.day > 31 || Number(time[1]) > 23 || Number(time[2]) > 59) {
            throw new Error("No hay fecha y hora válidas para generar el calendario.");
        }
        const start = new Date(parts.year, parts.month - 1, parts.day, Number(time[1]), Number(time[2]), 0, 0);
        if (start.getFullYear() !== parts.year || start.getMonth() !== parts.month - 1 || start.getDate() !== parts.day) {
            throw new Error("La fecha de la cita no es válida.");
        }
        return start;
    }

    function downloadFile(contents, mimeType, filename) {
        const blob = new Blob([contents], { type: mimeType });
        const url = URL.createObjectURL(blob);
        const anchor = document.createElement("a");
        anchor.href = url;
        anchor.download = filename;
        document.body.appendChild(anchor);
        anchor.click();
        anchor.remove();
        window.setTimeout(() => URL.revokeObjectURL(url), 1000);
    }

    window.descargarTicket = function () {
        const d = bookingData;
        if (!d) return;
        const lines = [
            "MARIAN MADRID SALÓN",
            "TICKET DE CONFIRMACIÓN DE CITA",
            "",
            "Servicio: " + d.tituloServicio,
            "ID de reserva: " + d.id,
            ...(d.idF2 ? ["ID de reserva Fase 2: " + d.idF2] : []),
            "Fecha: " + d.fecha,
            "Hora: " + d.hora,
            ...(d.esDual ? ["Hora Fase 2: " + (d.horaF2 || "Pendiente")] : []),
            "Duración: " + (d.duracion > 0 ? d.duracion + " min" : "No indicada"),
            "Complementos: " + d.extras,
            "Precio estimado: " + (d.precio > 0
                ? new Intl.NumberFormat("es-ES", { style: "currency", currency: "EUR" }).format(d.precio)
                : "Precio en salón"),
            ...(d.appointmentNote ? ["Comentarios: " + d.appointmentNote] : []),
            "",
            "La asignación del profesional la gestiona el salón."
        ];
        const filenameDate = d.fecha.replace(/[^0-9-]/g, "-");
        downloadFile(lines.join("\r\n"), "text/plain;charset=utf-8",
            "MarianMadrid_Ticket_" + filenameDate + ".txt");
    };

    window.descargarCalendario = function () {
        const d = bookingData;
        if (!d) return;
        let start;
        try {
            start = localDateTime(d);
        } catch (error) {
            window.alert(error.message);
            return;
        }
        const end = d.duracion > 0 ? new Date(start.getTime() + d.duracion * 60000) : null;
        const toIcsLocal = (date) => date.getFullYear() + pad2(date.getMonth() + 1) +
            pad2(date.getDate()) + "T" + pad2(date.getHours()) + pad2(date.getMinutes()) + "00";
        const nowUtc = new Date().toISOString().replace(/[-:]/g, "").replace(/\.\d{3}Z$/, "Z");
        const description = [
            "Cita: " + d.servicio,
            "ID de reserva: " + d.id,
            ...(d.idF2 ? ["ID de reserva Fase 2: " + d.idF2] : []),
            "Complementos: " + d.extras,
            ...(d.duracion > 0 ? ["Duración: " + d.duracion + " min"] : []),
            ...(d.appointmentNote ? ["Comentarios: " + d.appointmentNote] : [])
        ].join("\n");
        const lines = [
            "BEGIN:VCALENDAR",
            "VERSION:2.0",
            "PRODID:-//Marian Madrid//Booking Calendar//ES",
            "CALSCALE:GREGORIAN",
            "BEGIN:VEVENT",
            "UID:" + escapeIcs(d.id) + "@marianmadrid.es",
            "DTSTAMP:" + nowUtc,
            "DTSTART;TZID=Europe/Madrid:" + toIcsLocal(start),
            ...(end ? ["DTEND;TZID=Europe/Madrid:" + toIcsLocal(end)] : []),
            "SUMMARY:" + escapeIcs("Marian Madrid - " + d.tituloServicio),
            "DESCRIPTION:" + escapeIcs(description),
            "END:VEVENT",
            "END:VCALENDAR"
        ];
        const filenameDate = d.fecha.replace(/[^0-9]/g, "");
        downloadFile(lines.join("\r\n"), "text/calendar;charset=utf-8",
            "MarianMadrid_" + filenameDate + ".ics");
    };

    window.irAInicio = () => sendNavigation(NAV_HOME);

    window.addEventListener("message", (event) => {
        if (event.source !== window.parent) return;
        if (!PARENT_ORIGIN || event.origin !== PARENT_ORIGIN || !isTrustedOrigin(event.origin)) return;
        const message = event.data;
        if (!message || message.version !== MESSAGE_VERSION || typeof message.type !== "string") return;
        if (message.type === "MM_CONFIRMATION_CONTEXT") {
            const normalized = normalizePayload(message.payload);
            if (!normalized) return;
            bookingData = normalized;
            render(normalized);
        }
    });

    });

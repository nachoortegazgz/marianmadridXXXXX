        <h1 id="serviceTitle">Cargando servicio…</h1>

        </div>
        <div id="recommendationsSection" hidden>
          <h2 class="section-title">Recomendaciones</h2>
          <ul class="recommendations" id="recommendationsList"></ul>
        </div>
        <div class="total">

          <strong id="totalPrice">—</strong>
        </div>
        <button class="action" id="primaryAction" type="button" disabled>Preparando…</button>
        <p class="notice" id="statusNotice" role="status" aria-live="polite">Conectando con el servicio de reservas…</p>
        <p class="notice error" id="errorNotice" role="alert" hidden></p>

  <script>
    "use strict";

    const VERSION = 1;
    const TYPES = Object.freeze({
      READY: "MM_READY",
      CONTEXT: "MM_CONTEXT",
      BOOK: "MM_BOOK",
      NAV: "MM_NAV"
    });
    const MAX_ADDONS = 5;
    const state = {
      bridgeId: "",
      service: null,
      selectedAddons: new Set(),
      busy: false
    };
    const byId = (id) => document.getElementById(id);
    const text = (value, fallback = "") => {
      const valueText = String(value == null ? "" : value).trim();
      return valueText || fallback;
    };
    const number = (value) => {
      const parsed = Number(value);
      return Number.isFinite(parsed) ? parsed : 0;
    };
    const messageId = (prefix) =>
      prefix + "-" + Date.now().toString(36) + "-" +
      Math.random().toString(36).slice(2, 10);
    const money = (value, currency) => {
      const code = /^[A-Z]{3}$/.test(text(currency).toUpperCase())
        ? text(currency).toUpperCase()
        : "EUR";
      try {
        return new Intl.NumberFormat("es-ES", {
          style: "currency",
          currency: code
        }).format(Math.max(0, number(value)));
      } catch (_) {
        return Math.max(0, number(value)).toFixed(2).replace(".", ",") + " €";
      }
    };
    function send(type, payload = {}) {
      if (!window.parent || window.parent === window) return false;
      window.parent.postMessage({
        version: VERSION,
        type,
        messageId: messageId(type.toLowerCase()),
        bridgeId: state.bridgeId,
        payload
      }, "*");
      return true;
    }
    function safeImageUrl(value) {
      const raw = typeof value === "string"
        ? value.trim()
        : text(value && (value.url || value.src || value.image));
      if (/^https:\/\/static\.wixstatic\.com\/media\//i.test(raw)) {
        return raw;
      }
      const match = raw.match(/^wix:image:\/\/v1\/([^/#?]+)/i);
      return match
        ? "https://static.wixstatic.com/media/" + match[1]
        : "";
    }
    function getAddons(service) {
      const options = Array.isArray(service.addOnOptions)
        ? service.addOnOptions
        : [];
      const seen = new Set();
      return options.slice(0, MAX_ADDONS).map((item) => {
        if (!item || typeof item !== "object") return null;
        const id = text(item.addonId || item.addOnId || item.id);
        if (!id || seen.has(id)) return null;
        seen.add(id);
        return {
          id,
          name: text(item.name || item.title, "Complemento"),
          price: Math.max(0, number(item.price))
        };
      }).filter(Boolean);
    }
    function selectedAddonTotal() {
      return getAddons(state.service || {}).reduce((total, addon) =>
        total + (state.selectedAddons.has(addon.id) ? addon.price : 0), 0);
    }
    function showError(message) {
      byId("errorNotice").textContent = text(message, "No se pudo cargar el servicio.");
      byId("errorNotice").hidden = false;
      byId("statusNotice").textContent = "";
    }
    function updateTotal() {
      const service = state.service;
      if (!service) return;
      const total = Math.max(0, number(service.price)) + selectedAddonTotal();
      byId("totalPrice").textContent = money(total, service.currency);
      byId("servicePrice").textContent = money(service.price, service.currency);
      byId("totalLabel").textContent = service.catalogEntryType === "PRICING_PLAN"
        ? "Precio del bono"
        : "Precio estimado";
    }
    function renderAddons(service) {
      const container = byId("addonsList");
      container.replaceChildren();
      const options = getAddons(service);
      state.selectedAddons = new Set();
      byId("addonsSection").hidden = options.length === 0;
      options.forEach((addon) => {
        const label = document.createElement("label");
        label.className = "addon";
        const checkbox = document.createElement("input");
        checkbox.type = "checkbox";
        checkbox.value = addon.id;
        checkbox.addEventListener("change", () => {
          if (checkbox.checked) state.selectedAddons.add(addon.id);
          else state.selectedAddons.delete(addon.id);
          updateTotal();
        });
        const copy = document.createElement("span");
        copy.className = "addon-copy";
        const name = document.createElement("span");
        name.className = "addon-name";
        name.textContent = addon.name;
        const price = document.createElement("span");
        price.className = "addon-price";
        price.textContent = "+" + money(addon.price, service.currency);
        copy.append(name, price);
        label.append(checkbox, copy);
        container.appendChild(label);
      });
    }
    function renderRecommendations(value) {
      const container = byId("recommendationsList");
      container.replaceChildren();
      const items = Array.isArray(value)
        ? value.map((item) => text(item && (item.text || item.title || item.name) || item))
            .filter(Boolean).slice(0, 8)
        : [];
      byId("recommendationsSection").hidden = items.length === 0;
      items.forEach((item) => {
        const li = document.createElement("li");
        li.textContent = item;
        container.appendChild(li);
      });
    }
    function renderService(service) {
      if (!service || typeof service !== "object" || Array.isArray(service)) {
        showError("No llegaron los datos del servicio. Recarga la página.");
        return;
      }
      const type = text(service.catalogEntryType, "SERVICE").toUpperCase();
      if (type !== "SERVICE" && type !== "PRICING_PLAN") {
        showError("El tipo de entrada del catálogo no es compatible.");
        return;
      }
      state.service = service;
      byId("serviceTitle").textContent = text(service.title, "Servicio Marian Madrid");
      byId("serviceTagline").textContent = text(service.tagLine);
      byId("serviceDescription").textContent = text(service.description, "Consulta los detalles con nuestro equipo.");
      byId("serviceLocation").textContent = text(service.location, "Marian Madrid");
      byId("serviceLocation").hidden = false;
      const duration = number(service.totalDuration);
      byId("serviceDuration").textContent = duration > 0 ? duration + " min" : "";
      byId("serviceDuration").hidden = duration <= 0;
      const sessions = number(service.sessionCount);
      byId("serviceSessions").textContent = sessions > 0 ? sessions + (sessions === 1 ? " sesión" : " sesiones") : "";
      byId("serviceSessions").hidden = sessions <= 0;
      byId("eyebrow").textContent = type === "PRICING_PLAN"
        ? "Bono de sesiones"
        : "Marian Madrid · Peluquería y belleza";
      const imageUrl = safeImageUrl(service.mainMedia);
      byId("serviceImage").hidden = !imageUrl;
      byId("serviceImage").src = imageUrl;
      byId("serviceImage").alt = text(service.title, "Servicio Marian Madrid");
      byId("primaryAction").textContent = text(
        service.primaryActionLabel,
        type === "PRICING_PLAN" ? "Comprar bono" : "Reservar cita"
      );
      const unavailable = service.clientHidden === true;
      byId("primaryAction").disabled = unavailable;
      byId("statusNotice").textContent = unavailable
        ? "Este servicio no está disponible para reserva online."
        : "";
      renderAddons(service);
      renderRecommendations(service.recommendations);
      updateTotal();
      if (unavailable) return;
      byId("errorNotice").hidden = true;
    }
    function handleMessage(event) {
      if (event.source !== window.parent) return;
      const message = event.data;
      if (!message || typeof message !== "object" || message.version !== VERSION) return;
      if (message.type === TYPES.CONTEXT) {
        const incomingBridgeId = text(message.bridgeId);
        if (state.bridgeId && incomingBridgeId && state.bridgeId !== incomingBridgeId) return;
        state.bridgeId = incomingBridgeId;
        renderService(message.payload);
        return;
      }
      if (message.type === TYPES.BOOK) {
        const payload = message.payload && typeof message.payload === "object" ? message.payload : {};
        if (payload.status === "ERROR") {
          state.busy = false;
          byId("primaryAction").disabled = false;
          showError(payload.error && payload.error.message);
        } else if (payload.status === "SUCCESS") {
          state.busy = false;
          byId("primaryAction").disabled = true;
          byId("statusNotice").textContent = state.service && state.service.catalogEntryType === "PRICING_PLAN"
            ? "La solicitud de compra se ha enviado."
            : "Abriendo el calendario de reservas…";
        }
      }
    }
    function init() {
      byId("backButton").addEventListener("click", () => {
        send(TYPES.NAV, { target: "SERVICIOS" });
      });
      byId("primaryAction").addEventListener("click", () => {
        if (!state.service || state.busy || state.service.clientHidden === true) return;
        state.busy = true;
        byId("primaryAction").disabled = true;
        byId("statusNotice").textContent = state.service.catalogEntryType === "PRICING_PLAN"
          ? "Abriendo la compra segura…"
          : "Abriendo el calendario de reservas…";
        byId("errorNotice").hidden = true;
        send(TYPES.BOOK, {
          addOnIds: Array.from(state.selectedAddons).slice(0, MAX_ADDONS)
        });
      });
      window.addEventListener("message", handleMessage);
      send(TYPES.READY, { status: "INIT" });
    }
    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", init, { once: true });
    } else {
      init();
    }
  </script>
